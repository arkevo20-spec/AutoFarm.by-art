local function getRoot()

		return nil
	end

----------------------------------------------------
-- FIND RARITY
----------------------------------------------------

local function getRarity(instance)

	local current = instance

	for _ = 1, 10 do

		if not current then
			break
		end

		-- Attribute: Rarity
		local rarity =
			current:GetAttribute("Rarity")

		if typeof(rarity) == "string"
			and rarity ~= "" then
			return rarity
		end

		-- Attribute: Tier
		local tier =
			current:GetAttribute("Tier")

		if typeof(tier) == "string"
			and tier ~= "" then
			return tier
		end

		-- StringValue: Rarity
		local rarityValue =
			current:FindFirstChild("Rarity")

		if rarityValue
			and rarityValue:IsA("StringValue") then

			return rarityValue.Value
		end

		-- StringValue: Tier
		local tierValue =
			current:FindFirstChild("Tier")

		if tierValue
			and tierValue:IsA("StringValue") then

			return tierValue.Value
		end

		current = current.Parent
	end

	return nil
end

----------------------------------------------------
-- NORMALIZE RARITY
----------------------------------------------------

local function normalizeRarity(rarity)

	if not rarity then
		return nil
	end

	local lower =
		string.lower(tostring(rarity))

	for _, allowed in ipairs(rarities) do

		if string.lower(allowed) == lower then
			return allowed
		end
	end

	return tostring(rarity)
end

----------------------------------------------------
-- GET PART
----------------------------------------------------

local function getPartFromInstance(instance)

	if instance:IsA("BasePart") then
		return instance
	end

	if instance:IsA("Attachment") then

		if instance.Parent
			and instance.Parent:IsA("BasePart") then

			return instance.Parent
		end
	end

	if instance:IsA("Model") then

		if instance.PrimaryPart then
			return instance.PrimaryPart
		end

		local part =
			instance:FindFirstChildWhichIsA(
				"BasePart",
				true
			)

		if part then
			return part
		end
	end

	local current = instance

	for _ = 1, 8 do

		if not current then
			break
		end

		if current:IsA("BasePart") then
			return current
		end

		if current:IsA("Model") then

			if current.PrimaryPart then
				return current.PrimaryPart
			end

			local part =
				current:FindFirstChildWhichIsA(
					"BasePart",
					true
				)

			if part then
				return part
			end
		end

		current = current.Parent
	end

	return nil
end

----------------------------------------------------
-- STRING CONTAINS
----------------------------------------------------

local function containsKeyword(text, keywords)

	text = string.lower(tostring(text))

	for _, keyword in ipairs(keywords) do

		if string.find(
			text,
			string.lower(keyword),
			1,
			true
		) then

			return true
		end
	end

	return false
end

----------------------------------------------------
-- GET BUTTON TEXT
----------------------------------------------------

local function getButtonText(button)

	local text = ""

	pcall(function()
		text = text .. " " .. tostring(button.Name)
	end)

	pcall(function()
		if button:IsA("TextButton") then
			text = text .. " " .. tostring(button.Text)
		end
	end)

	return string.lower(text)
end

----------------------------------------------------
-- CLICK MATCHING GUI BUTTON
----------------------------------------------------

local function clickGuiButton(keywords)

	local found = nil

	for _, gui in ipairs(playerGui:GetDescendants()) do

		if gui:IsA("GuiButton") then

			local visible = true

			pcall(function()
				visible = gui.Visible
			end)

			if visible then

				local text =
					getButtonText(gui)

				if containsKeyword(
					text,
					keywords
				) then

					found = gui
					break
				end
			end
		end
	end

	if not found then
		return false
	end

	-- Try normal GuiButton activation first
	pcall(function()
		found:Activate()
	end)

	-- Wave fallback
	pcall(function()
		if found:IsA("TextButton")
			or found:IsA("ImageButton") then

			firesignal(
				found.MouseButton1Click
			)
		end
	end)

	return true
end

----------------------------------------------------
-- BAG PARSER
----------------------------------------------------

local function parseBagText(text)

	if not text then
		return nil, nil
	end

	local current, maximum =
		string.match(
			text,
			"(%d+)%s*/%s*(%d+)"
		)

	if current and maximum then

		return tonumber(current),
			tonumber(maximum)
	end

	return nil, nil
end

----------------------------------------------------
-- CHECK BAG
----------------------------------------------------

local function checkBag()

	local bestCurrent = nil
	local bestMax = nil

	for _, gui in ipairs(playerGui:GetDescendants()) do

		if gui:IsA("TextLabel")
			or gui:IsA("TextButton") then

			local text =
				tostring(gui.Text)

			local current, maximum =
				parseBagText(text)

			if current and maximum then

				local relevant = false

				local obj = gui

				for _ = 1, 5 do

					if not obj then
						break
					end

					if containsKeyword(
						obj.Name,
						Config.BagKeywords
					) then

						relevant = true
						break
					end

					obj = obj.Parent
				end

				if relevant then

					bestCurrent = current
					bestMax = maximum
					break
				end
			end
		end
	end

	if bestCurrent and bestMax then

		bagStatus.Text =
			"Bag: " ..
			bestCurrent ..
			" / " ..
			bestMax

		return bestCurrent >= bestMax
	end

	bagStatus.Text =
		"Bag: Unknown"

	return false
end

local function returnAndSell()

	status.Text =
		"Status: Returning to base..."

	local baseClicked =
		clickGuiButton(
			Config.BaseKeywords
		)

	if baseClicked then
		task.wait(0.7)
	else
		warn(
			"[WaveAutoFarm] Base button not found"
		)
	end

	if Config.AutoSell then

		status.Text =
			"Status: Selling..."

		task.wait(0.3)

		local sellClicked =
			clickGuiButton(
				Config.SellKeywords
			)

		if not sellClicked then

			warn(
				"[WaveAutoFarm] Sell button not found"
			)
		end
	end

	task.wait(0.8)
end

local function getTargetContainer(instance)

	local current = instance

	for _ = 1, 10 do

			break

			return current
		end

		current = current.Parent

	return instance.Parent
end

----------------------------------------------------
-- FIND TARGETS
----------------------------------------------------

local function findBestTarget()

	local root = getRoot()

	if not root then
		return nil
	end

	local nearest = nil
	local nearestDistance = math.huge

	for _, object in ipairs(workspace:GetDescendants()) do

		------------------------------------------------
		-- PROXIMITY PROMPT
		------------------------------------------------

		if object:IsA("ProximityPrompt") then

			if object.Enabled ~= false then

				local rarity =
					normalizeRarity(
						getRarity(object)
					)

				if rarity
					and allowedRarity[rarity] then

					local part =
						getPartFromInstance(object)

					if part then

						local distance =
							(
								root.Position
								-
								part.Position
							).Magnitude

						if distance < nearestDistance then

							nearestDistance = distance

							nearest = {
								Type = "Prompt",
								Object = object,
								Part = part,
								Rarity = rarity,
								Distance = distance,
								Container =
									getTargetContainer(object)
							}
						end
					end
				end
			end

		------------------------------------------------
		-- CLICK DETECTOR
		------------------------------------------------

		elseif object:IsA("ClickDetector") then

			local rarity =
				normalizeRarity(
					getRarity(object)
				)

			if rarity
				and allowedRarity[rarity] then

				local part =
					getPartFromInstance(object)

				if part then

					local distance =
						(
							root.Position
							-
							part.Position
						).Magnitude

					if distance < nearestDistance then

						nearestDistance = distance

						nearest = {
							Type = "Click",
							Object = object,
							Part = part,
							Rarity = rarity,
							Distance = distance,
							Container =
								getTargetContainer(object)
						}
					end
				end
			end
		end
	end

	return nearest
end

----------------------------------------------------
-- COLLECT TARGET
----------------------------------------------------

local function collectTarget(target)

	if not target then
		return false
	end

	local root = getRoot()

	if not root then
		return false
	end

	if not target.Part
		or not target.Part.Parent then
		return false
	end

	targetStatus.Text =
		"Target: " ..
		tostring(target.Rarity)

	------------------------------------------------
	-- TELEPORT
	------------------------------------------------

	pcall(function()

		root.CFrame =
			target.Part.CFrame
			+
			Config.TeleportOffset
	end)

	task.wait(
		Config.TeleportDelay
	)

	------------------------------------------------
	-- PROMPT
	------------------------------------------------

	if target.Type == "Prompt" then

		pcall(function()
			fireproximityprompt(
				target.Object
			)
		end)

		return true
	end

	------------------------------------------------
	-- CLICK DETECTOR
	------------------------------------------------

	if target.Type == "Click" then

		pcall(function()

			fireclickdetector(
				target.Object
			)

		end)

		return true
	end

	return false
end

----------------------------------------------------
-- AUTO FARM
----------------------------------------------------

local function startFarm()

	farmToken += 1

	local token = farmToken

	task.spawn(function()

		while isRunning
			and token == farmToken do

			------------------------------------------------
			-- BAG CHECK
			------------------------------------------------

			if Config.AutoSell
				and checkBag() then

				returnAndSell()

			else

				------------------------------------------------
				-- FIND TARGET
				------------------------------------------------

				local target =
					findBestTarget()

				if target then

					status.Text =
						"Status: Collecting " ..
						tostring(target.Rarity)

					collectTarget(
						target
					)

				else

					status.Text =
						"Status: No matching target"

					targetStatus.Text =
						"Target: None"

					task.wait(
						Config.ScanDelay
					)
				end
			end

			task.wait(
				Config.CollectDelay
			)
		end
	end)
end
